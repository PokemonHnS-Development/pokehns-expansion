# Cold boot, CONTINUE, walk OUT of the Pokemon Center first, walk around the town, then
# open the Pokedex, the party, the bag and the trainer card from the start menu. Every

leavecenter
30 -

walk LEFT 3
walk RIGHT 6

120 -

shot v04b_pokedex
waitctl 1200 B
startmenu 1                   # POKEMON (party)
90 -
shot v05_party
waitctl 1200 B
startmenu 2                   # BAG
90 -
shot v06_bag

120 -
shot v07_card_front

shot v07b_card_back

120 -
