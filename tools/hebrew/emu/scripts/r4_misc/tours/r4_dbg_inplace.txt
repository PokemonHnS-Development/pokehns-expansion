# Experiment for the suggested fix (Hebrew ROM only; rtlmode is a no-op on the English ROM): the
# debug menu's windows switched to WINDOW_RTL_IN_PLACE after they are created, as
# SetWindowRtlMode(windowId, WINDOW_RTL_IN_PLACE) after each AddWindow in src/debug.c would do.
# In-place mode keeps a segment with no Hebrew in its English layout (src/text.c RtlPrepareGlyph,
# rtlSegmentLtr) and mirrors a Hebrew segment inside its own extent. A key press reprints the window.
start saves/r4_misc/r4_menuA_base.sav
include ../../continue.txt

# main list: UP wraps to the last item (the page scrolls and is reprinted), DOWN wraps back
debugmenu
rtlmode DebugTask_HandleMenuInput_General 1 1
tap UP 6
tap DOWN 6
waitquiet
snap dbg_inplace_main
waitctl 600 B

# Flags & Vars (colour codes inside the rows)
dbg 7
rtlmode DebugTask_HandleMenuInput_General 1 1
tap UP 6
tap DOWN 6
waitquiet
snap dbg_inplace_flags
waitctl 600 B

# Give item XYZ (Hebrew item name, icon sprite, digit indicator)
dbg 3 0
waitquiet
rtlmode DebugAction_Give_Item_SelectId 2 1
tap UP 6
waitquiet
snap dbg_inplace_give_item
press A
rtlmode DebugAction_Give_Item_SelectQuantity 2 1
tap UP 6
waitquiet
snap dbg_inplace_give_qty
tap B 6
waitctl 600 B

# Pokemon (Basic): species

waitquiet

tap UP 6
waitquiet

waitctl 600 B

tap B 6
