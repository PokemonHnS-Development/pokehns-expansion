# Every Easy Chat screen type Heart & Soul can reach, opened on the Hebrew build with the game's own
# script (Common_ShowEasyChatScreen, data/event_scripts.s:1187) and the type in VAR_0x8004, the way
# the scripts that reach them set it (the feelings man's BATTLE_START/WON/LOST, the contest and
# Battle Tower reporters' interviews, the apprentice's win speech, the mart questionnaire; the Day
# Care INTERVIEW is ec18_ui.txt and MAIL is ec18_mail.txt). In each: heapstat with the screen open,
# the alphabet keyboard, the longest list (mem) paged to its end, the group keyboard, one word put
# into the first slot, and the YES/NO box of the exit prompt (the contest interview and the
# apprentice cannot be left: B shows 'you cannot quit here', so they are confirmed with OK -> YES);
# then back to the field. BATTLE_WON gets six words and the OK prompt, the C3 case (its YES/NO box
# sat over the third row's first word).
start saves/ladder6_postgame.sav
include ../continue.txt
ev setflag FLAG_SYS_GAME_CLEAR; setvar VAR_HAS_ENTERED_BATTLE_FRONTIER, 1; end

# EASY_CHAT_TYPE_BATTLE_START
evstart setvar VAR_0x8004, EASY_CHAT_TYPE_BATTLE_START; call Common_ShowEasyChatScreen; end
waitcb2x CB2_EasyChatScreen 900
ecwait
heapstat start
ecstate
snap ty_start_screen
ec A
ecstate
snap ty_start_kbd
ec SELECT
ecletter 2 0
snap ty_start_abc
ec A
eclist ty_start_mem 80
ec B
ec SELECT
ec B
ecputword 0 0 EC_GROUP_FEELINGS EC_WORD_HAPPY ty_start_put
snap ty_start_1word
ec B
ecstate
snap ty_start_quit
ec UP
tap A 0
waitcb2x CB2_Overworld 900
waitctl 900
