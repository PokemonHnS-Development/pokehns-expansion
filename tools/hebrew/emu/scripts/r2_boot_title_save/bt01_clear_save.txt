# Title B+SELECT+UP: clear-save screen (src/clear_save_data_screen.c; strings.c:1233-1234).
# NO first (soft reset), then YES: "Clearing data... please wait" while ClearSaveData erases the
# flash, then the reset and the main menu with no save (NEW GAME / OPTION).

tap UP 4

regionwatch bt01_03_clearing 900 120 152 CB2_InitCopyrightScreenAfterBootup
include _mainmenu.txt
snap bt01_04_mainmenu_nosave
