# battle 1: box 1 has room
evstart setwildbattle SPECIES_PAWMI, 5; dowildbattle; end
waitbattle
bwalk heid_rt_b1_intro input:HandleInputChooseAction
baction BAG
waitcb2x CB2_BagMenuRun 900
waittaskx Task_BagMenu_HandleInput 900
waitquiet
bagpocket 2
bagseek ITEM_MASTER_BALL
waitquiet
tap A 4                                   # USE / CANCEL
waitquiet
tap A 4                                   # USE: the throw
bwalk heid_rt_b1_throw task:Task_HandleCaughtMonPageInput|prompt:Cmd_trygivecaughtmonnick
snap heid_rt_b1_after_throw
tap A 4                                   # the Pokedex page
bwalk heid_rt_b1_after prompt:Cmd_trygivecaughtmonnick
snap heid_rt_b1_nick_prompt
tap DOWN 4                                # no nickname
tap A 4
bwalk heid_rt_b1_swapq prompt:Cmd_givecaughtmon
snap heid_rt_b1_swap_prompt
tap DOWN 4                                # NO: the caught Pokemon goes to the PC
tap A 4
bwalk heid_rt_b1_sent free
snap heid_rt_b1_field
