# r10 names: the ability pop-up (FONT_SMALL, 76 px budget) with the widest ability names, on the
# player's side (Porygon2's Trace copies the foe's ability and the pop-up shows the copied name)
# and on the foe's side (a foe Porygon2 traces the player's Pidgeot's TANGLED FEET); the FIGHT
# menu (64 px FONT_NARROW slots) with the widest learnable move names, cursor on each slot.
start saves/ladder1_newgame.sav
include ../_battle_base.txt
ev createmon 0, 0, SPECIES_PORYGON2, 60, abilityNum=0, move1=MOVE_SUNSTEEL_STRIKE, move2=MOVE_DARKEST_LARIAT, move3=MOVE_MALIGNANT_CHAIN, move4=MOVE_TRICK_OR_TREAT; createmon 0, 1, SPECIES_PIDGEOT, 60, abilityNum=1, move1=MOVE_CHILLY_RECEPTION, move2=MOVE_TRIPLE_DIVE, move3=MOVE_CLANGING_SCALES, move4=MOVE_PARABOLIC_CHARGE; special CalculatePlayerPartyCount; end
pinrng
# ---- foe SPECIES_PIDGEOT (abilityNum 1), traced by the player's Porygon2
evstart createmon 1, 0, SPECIES_PIDGEOT, 5, abilityNum=1; dowildbattle; end
waitbattle
bwalk n_bab_1_tangled input:HandleInputChooseAction
snap n_bab_1_tangled_action
baction FIGHT
waitbinput HandleInputChooseMove
waitquiet
snap n_bab_1_tangled_fight_0
bmove 1 nopress
waitquiet
snap n_bab_1_tangled_fight_1
bmove 2 nopress
waitquiet
snap n_bab_1_tangled_fight_2
bmove 3 nopress
waitquiet
snap n_bab_1_tangled_fight_3
tap B 4
waitbinput HandleInputChooseAction
baction RUN
bwalk n_bab_1_tangled_run free
waitctl 600
