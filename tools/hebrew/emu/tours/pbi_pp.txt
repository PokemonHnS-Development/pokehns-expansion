# PP items from the bag (src/party_menu.c ItemUseCB_PPUp :5574, ItemUseCB_PPRecovery :5486,
# ShowMoveSelectWindow :5448): "Boost PP of which move?" and "Restore PP of which move?" with
# the move list window, "PP increased", "PP was restored" (Ether on one move, Elixir on all),
# and "It won't have any effect." on a move at full PP.
# Two wild battles against a 1-HP Rattata first spend 1 PP of Typhlosion's moves 1 and 2
# (EMBER, TACKLE set here), so there is PP to restore.
start saves/menuA_base.sav
include ../scripts/continue.txt
ev setmonmove 0, 0, MOVE_EMBER; setmonmove 0, 1, MOVE_TACKLE; additem ITEM_PP_UP, 3; additem ITEM_ETHER, 3; additem ITEM_ELIXIR, 3; end

pinrng
ev setwildbattle SPECIES_RATTATA, 2; end
evstart dowildbattle; end
waitbattle
sethp enemy 0 1
bwalk pbi_pp_b1_intro input:HandleInputChooseAction
baction FIGHT
bmove 0
bwalk pbi_pp_b1_end free
ev setwildbattle SPECIES_RATTATA, 2; end
evstart dowildbattle; end
waitbattle
sethp enemy 0 1
bwalk pbi_pp_b2_intro input:HandleInputChooseAction
baction FIGHT
bmove 1
bwalk pbi_pp_b2_end free
unpinrng

# MEDICINE: 0 POTION .. 5 RARE CANDY, then 6 PP UP, 7 ETHER, 8 ELIXIR (added above)
