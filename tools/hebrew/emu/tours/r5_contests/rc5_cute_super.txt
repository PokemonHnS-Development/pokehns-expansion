# A Super-rank CUTE contest from the H&S contest lobby (MAP_LILYCOVE_CITY_CONTEST_LOBBY_HNS).
# Party slot 0 becomes a level-50 PIKACHU made by this ROM (so its nickname is the ROM's species name)
# with the moves below; the appeals are the digit string 01233 (the move-list index for each round):
#   0 BABY_DOLL_EYES (CUTE, CONTEST_EFFECT_NEXT_APPEAL_EARLIER, src/data/moves_info.h)
#   1 AFTER_YOU (CUTE, CONTEST_EFFECT_NEXT_APPEAL_LATER, src/data/moves_info.h)
#   2 TICKLE (CUTE, CONTEST_EFFECT_WORSEN_CONDITION_OF_PREV_MONS, src/data/moves_info.h)
#   3 DEFENSE_CURL (CUTE, CONTEST_EFFECT_AVOID_STARTLE_ONCE, src/data/moves_info.h)
# Baby-Doll Eyes (earlier), After You (later), Tickle (worsen condition), Defense Curl twice (settle down; repeated appeal).
# Entering Super rank needs the CUTE ribbon of the rank below (GetContestEntryEligibility); they are
# given rank by rank with GiveMonContestRibbon on gContestMonPartyIndex 0.
# Round 1 points are set to 600 once the contest screen starts (gContestMonRound1Points[3], the
# player is contestant 3), so the player wins and gets the ribbon (contest_hall.inc ReceiveContestRibbon).
# gRngValue is pinned from the lobby on (pinrng), so the opponents, the AI's appeals and the
# audience are the same on the Hebrew and the English ROM (the shots pair up by name).
start saves/ladder6_postgame.sav
include ../../scripts/continue.txt
ev createmon 0, 0, SPECIES_PIKACHU, 50, move1=MOVE_BABY_DOLL_EYES, move2=MOVE_AFTER_YOU, move3=MOVE_TICKLE, move4=MOVE_DEFENSE_CURL; additem ITEM_POKEBLOCK_CASE, 1; end
w8 @gContestMonPartyIndex@ 0
ev setvar VAR_CONTEST_CATEGORY, 2; setvar VAR_CONTEST_RANK, 0; special GiveMonContestRibbon; end

tap A 4
