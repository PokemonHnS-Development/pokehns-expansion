# A Hyper-rank SMART contest from the H&S contest lobby (MAP_LILYCOVE_CITY_CONTEST_LOBBY_HNS).
# Party slot 0 becomes a level-50 SLOWKING made by this ROM (so its nickname is the ROM's species name)
# with the moves below; the appeals are the digit string 01233 (the move-list index for each round):
#   0 FLATTER (SMART, CONTEST_EFFECT_MAKE_FOLLOWING_MONS_NERVOUS, src/data/moves_info.h)
#   1 STRING_SHOT (SMART, CONTEST_EFFECT_STARTLE_PREV_MON, src/data/moves_info.h)
#   2 SKILL_SWAP (SMART, CONTEST_EFFECT_APPEAL_AS_GOOD_AS_PREV_ONES, src/data/moves_info.h)
#   3 MAGIC_ROOM (SMART, CONTEST_EFFECT_DONT_EXCITE_AUDIENCE, src/data/moves_info.h)
# Flatter (nervous), String Shot (startle the one before), Skill Swap (as good as the ones before), Magic Room twice (don't excite the audience; repeated).
# Entering Hyper rank needs the SMART ribbon of the rank below (GetContestEntryEligibility); they are
# given rank by rank with GiveMonContestRibbon on gContestMonPartyIndex 0.
# Round 1 points are set to 600 once the contest screen starts (gContestMonRound1Points[3], the
# player is contestant 3), so the player wins and gets the ribbon (contest_hall.inc ReceiveContestRibbon).
# gRngValue is pinned from the lobby on (pinrng), so the opponents, the AI's appeals and the
# audience are the same on the Hebrew and the English ROM (the shots pair up by name).
start saves/ladder6_postgame.sav
include ../../scripts/continue.txt
ev createmon 0, 0, SPECIES_SLOWKING, 50, move1=MOVE_FLATTER, move2=MOVE_STRING_SHOT, move3=MOVE_SKILL_SWAP, move4=MOVE_MAGIC_ROOM; additem ITEM_POKEBLOCK_CASE, 1; end
w8 @gContestMonPartyIndex@ 0
ev setvar VAR_CONTEST_CATEGORY, 3; setvar VAR_CONTEST_RANK, 0; special GiveMonContestRibbon; setvar VAR_CONTEST_RANK, 1; special GiveMonContestRibbon; end
warpto MAP_LILYCOVE_CITY_CONTEST_LOBBY_HNS 14 3
waitnotask Task_MapNamePopUpWindow 600
pinrng

tap A 4
