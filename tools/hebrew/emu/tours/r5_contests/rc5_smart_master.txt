# A Master-rank SMART contest from the H&S contest lobby (MAP_LILYCOVE_CITY_CONTEST_LOBBY_HNS).
# Party slot 0 becomes a level-50 PORYGON_Z made by this ROM (so its nickname is the ROM's species name)
# with the moves below; the appeals are the digit string 12300 (the move-list index for each round):
#   0 FLAMETHROWER (BEAUTY, CONTEST_EFFECT_HIGHLY_APPEALING, src/data/moves_info.h)
#   1 DISABLE (SMART, CONTEST_EFFECT_MAKE_FOLLOWING_MONS_NERVOUS, src/data/moves_info.h)
#   2 SKETCH (SMART, CONTEST_EFFECT_APPEAL_AS_GOOD_AS_PREV_ONE, src/data/moves_info.h)
#   3 EERIE_SPELL (SMART, CONTEST_EFFECT_BETTER_WHEN_LATER, src/data/moves_info.h)
# Disable (nervous), Sketch (as good as the one before), Eerie Spell (better when later), Flamethrower twice (a BEAUTY move: didn't go over well; repeated).
# Entering Master rank needs the SMART ribbon of the rank below (GetContestEntryEligibility); they are
# given rank by rank with GiveMonContestRibbon on gContestMonPartyIndex 0.
# Round 1 points are set to 600 once the contest screen starts (gContestMonRound1Points[3], the
# player is contestant 3), so the player wins and gets the ribbon (contest_hall.inc ReceiveContestRibbon).
# gRngValue is pinned from the lobby on (pinrng), so the opponents, the AI's appeals and the
# audience are the same on the Hebrew and the English ROM (the shots pair up by name).
start saves/ladder6_postgame.sav
include ../../scripts/continue.txt
ev createmon 0, 0, SPECIES_PORYGON_Z, 50, move1=MOVE_FLAMETHROWER, move2=MOVE_DISABLE, move3=MOVE_SKETCH, move4=MOVE_EERIE_SPELL; additem ITEM_POKEBLOCK_CASE, 1; end
w8 @gContestMonPartyIndex@ 0
ev setvar VAR_CONTEST_CATEGORY, 3; setvar VAR_CONTEST_RANK, 0; special GiveMonContestRibbon; setvar VAR_CONTEST_RANK, 1; special GiveMonContestRibbon; setvar VAR_CONTEST_RANK, 2; special GiveMonContestRibbon; end
warpto MAP_LILYCOVE_CITY_CONTEST_LOBBY_HNS 14 3
