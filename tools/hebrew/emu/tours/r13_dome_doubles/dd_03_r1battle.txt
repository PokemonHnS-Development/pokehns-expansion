# r13 Dome doubles, round 1: READY TO START, the pick of 2, the battle room, the double battle
# with every menu shown once (action menus of both battlers, the in-battle party menu and its
# 'already in battle' line, the real move menu, a single-target move's target cursor on both
# foes, a spread move's target display), the win, the pre-battle room, the static tree, the
# round-2 menu. Saves state r2menu.
start saves/r13/dome_doubles/base.sav
dstate load r1ready
pick 2
drivex dd_r1pre task:Task_HandleChooseMonInput 30 9000
pfx dd_r1
include _dd_pick2.txt

waitquiet
snap dd_r1b_party
seek 8 @gPartyMenu+9@ 1 DOWN
tap A 0
waittask Task_HandleSelectionMenuInput 120
waitquiet
snap dd_r1b_party_menu
pick 0
30 -
waitquiet
snap dd_r1b_party_inbattle
tap A 0
waittask Task_HandleChooseMonInput 300
waitquiet
tap B 0
waitbinput HandleInputChooseAction 1200
# the real move menu, then Ice Shard (single target) for battler 0
bactionb 0 FIGHT
waitbinputb 0 HandleInputChooseMove
snap dd_r1b_move0_real
tap B 0
waitbinputb 0 HandleInputChooseAction
bmove0 0 MOVE_ICE_SHARD
bhp4 1 max max

waitbinputb 0 HandleInputChooseMove
snap dd_r1b_move0

waitbinputb 0 HandleInputChooseTarget

waitquiet
snap dd_r1b_target0_b
tap LEFT 4
waitquiet
snap dd_r1b_target0_c

waitbinputb 2 HandleInputChooseAction

waitbinputb 2 HandleInputChooseMove

tap B 0
waitbinputb 2 HandleInputChooseAction

waitbinputb 2 HandleInputChooseMove

tap A 4

waitquiet
120 -
